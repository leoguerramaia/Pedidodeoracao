Script SQL das Tabelas (Execute no SQL Editor do Supabase caso ainda não tenha criado): 
-- No painel do Supabase, abra "SQL Editor" -> "New Query", cole tudo abaixo e clique em "Run":
create table if not exists public.usuarios (
  id text primary key,
  username text,
  nome text not null,
  sobrenome text not null,
  sexo text not null,
  telefone text not null,
  email text not null,
  cep text,
  rua text,
  bairro text,
  cidade text,
  estado text,
  numero text,
  complemento text,
  foto_perfil text,
  senha text not null,
  is_admin boolean default false,
  must_change_password boolean default false,
  reset_requested boolean default false,
  created_at timestamptz default now()
);

create table if not exists public.pedidos_oracao (
  id text primary key,
  user_id text,
  nome text not null,
  sobrenome text not null,
  sexo text not null,
  pedido text not null,
  created_at timestamptz default now()
);

alter table public.pedidos_oracao drop constraint if exists pedidos_oracao_user_id_fkey;
alter table public.usuarios drop constraint if exists usuarios_username_key;

alter table public.usuarios enable row level security;
alter table public.pedidos_oracao enable row level security;

drop policy if exists "Acesso publico usuarios" on public.usuarios;
create policy "Acesso publico usuarios" on public.usuarios for all to public using (true) with check (true);

drop policy if exists "Acesso publico pedidos" on public.pedidos_oracao;
create policy "Acesso publico pedidos" on public.pedidos_oracao for all to public using (true) with check (true);

grant usage on schema public to anon, authenticated;
grant all on table public.usuarios to anon, authenticated, service_role;
grant all on table public.pedidos_oracao to anon, authenticated, service_role;
