Conexão Supabase & Banco de Dados Online

Configure sua URL e Anon Key do Supabase para sincronizar na nuvem, ou use o armazenamento persistente integrado.

Script SQL para o SQL Editor do Supabase: 
-- Execute este script no SQL Editor do seu projeto Supabase:
create table if not exists public.usuarios (
  id text primary key,
  username text unique,
  nome text not null,
  sobrenome text not null,
  sexo text not null,
  telefone text not null,
  email text unique not null,
  cep text,
  rua text,
  bairro text,
  cidade text,
  estado text,
  numero text,
  complemento text,
  foto_perfil text,
  senha text not null,
  is_admin boolean default false,
  must_change_password boolean default false,
  reset_requested boolean default false,
  created_at timestamptz default now()
);

create table if not exists public.pedidos_oracao (
  id text primary key,
  user_id text references public.usuarios(id) on delete cascade,
  nome text not null,
  sobrenome text not null,
  sexo text not null,
  pedido text not null,
  created_at timestamptz default now()
);

alter table public.usuarios enable row level security;
alter table public.pedidos_oracao enable row level security;

create policy "Acesso publico usuarios" on public.usuarios for all using (true) with check (true);
create policy "Acesso publico pedidos" on public.pedidos_oracao for all using (true) with check (true);
